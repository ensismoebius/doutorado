##
## Optimizer unit tests (GoogleTest)
##
## Builds:
## - `optimizers_gtest`: tests for optimizer implementations (Adam/SGD variants).
##
## Notes:
## - Linked against `optimizers` + `tensor` and uses xtensor/OpenMP via
##   `configure_xtensor_parallel_target`.
##
add_executable(optimizers_gtest 
    optimizers_gtest.cpp
    state_io_gtest.cpp
)

target_link_libraries(optimizers_gtest 
    PRIVATE 
        google_test 
         xtensor xtensor-blas 
        optimizers
        tensor
)

target_include_directories(optimizers_gtest 
    PRIVATE 
        ${CMAKE_SOURCE_DIR}/src
)

include(GoogleTest)
nn_gtest_discover_tests(optimizers_gtest)
configure_xtensor_parallel_target(optimizers_gtest)

## optimizer_parity_gtest: ground-truth parity for every optimizer against its reference
## implementation (PyTorch's own for Adam/AdamW/SGD; the authors' lion-pytorch and
## schedulefree packages for the rest). Loads committed fixtures
## (fixtures/optimizer_refs.npz, generated by scripts/testing/gen_optimizer_refs.py), so
## no Python/torch is needed at build or CI time.
add_executable(optimizer_parity_gtest
    optimizer_parity_gtest.cpp
)

target_link_libraries(optimizer_parity_gtest
    PRIVATE
        google_test
        xtensor xtensor-blas
        optimizers
        tensor
        cnpy-static
        z
)

target_include_directories(optimizer_parity_gtest
    PRIVATE
        ${cnpy_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/src
)

set_target_properties(optimizer_parity_gtest PROPERTIES CXX_CLANG_TIDY "")

nn_gtest_discover_tests(optimizer_parity_gtest)
configure_xtensor_parallel_target(optimizer_parity_gtest)

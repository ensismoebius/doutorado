##
## Core layers unit tests (GoogleTest)
##
## Builds:
## - `layers_gtest`: broad coverage across dense layers and SNN components.
##
## Notes:
## - clang-tidy is disabled for this binary (test convenience; avoids noisy
##   diagnostics in third-party include stacks).
##
add_executable(layers_gtest
    layers_gtest.cpp
)

target_link_libraries(layers_gtest 
    PRIVATE 
        google_test 
         xtensor xtensor-blas 
        layers
        tensor
)

target_include_directories(
  layers_gtest 
    PRIVATE 
         ${cnpy_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/src
)

set_target_properties(layers_gtest PROPERTIES CXX_CLANG_TIDY "")

include(GoogleTest)
nn_gtest_discover_tests(layers_gtest)
configure_xtensor_parallel_target(layers_gtest)

##
## micro_network_parity_gtest: ground-truth parity for whole micro-NETWORKS (ANN / SNN /
## LSTM) against PyTorch + snnTorch, complementing pytorch_parity_gtest's per-layer checks.
## Layers can each be right while the network built from them is wrong (gradient chaining,
## state leaking across sequences, gate order only visible once composed). Loads committed
## fixtures (fixtures/micro_network_refs.npz, generated by
## scripts/testing/gen_micro_network_refs.py), so no Python/torch is needed at build or CI
## time.
##
add_executable(micro_network_parity_gtest
    micro_network_parity_gtest.cpp
)

target_link_libraries(micro_network_parity_gtest
    PRIVATE
        google_test
        xtensor xtensor-blas
        layers
        tensor
        cnpy-static
        z
)

target_include_directories(micro_network_parity_gtest
    PRIVATE
        ${cnpy_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/src
)

set_target_properties(micro_network_parity_gtest PROPERTIES CXX_CLANG_TIDY "")

nn_gtest_discover_tests(micro_network_parity_gtest)
configure_xtensor_parallel_target(micro_network_parity_gtest)

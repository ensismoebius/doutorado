/* WAVELETS : 23 wavelets-mae (haar, daub4, ..., daub46)         */
/* escalas  : voz -> {lfcc, mel, bark}; EEG -> {lfcc}             */
/* CATEG.   : 1 = seis descritores por grupo; 2 = cepstral       */
/* Cada configuracao roda 3 vezes; d_pen e a media das 3.        */
/* Retorna o vencedor (menor d_pen) e o melhor de cada familia.  */

Selecao seleciona_vencedor(Sinal s)
{
    Lista configs = vazia();

    /* 1) manual: wavelet x escala x categoria                    */
    for (int w = 0; w < 23; w++)
        for (int e = 0; e < n_escalas(s); e++)
            for (int c = 1; c <= 2; c++)
                adiciona(&configs, manual(WAVELETS[w],
                                          escala(s, e), c));

    /* 2) autoencoders: 9 SNN-AE (3 tamanhos x 3 codificacoes)    */
    /*    e 3 ANN-AE (3 tamanhos), sobre o espectro do sinal      */
    adiciona_autoencoders(&configs);
    /* 3) PCA com k = 8, 16 e 32, sobre a mesma entrada           */
    adiciona_pca(&configs);

    Selecao sel = { .vencedor.d_pen = INFINITO };
    for (int i = 0; i < configs.n; i++) {
        double d_pen = 0.0;
        for (int r = 0; r < 3; r++) {          /* 3 repeticoes   */
            double v[MAX_CARACT];
            int m = extrai(s, configs.item[i], r, v);
            /* alpha, beta -> G1, G2 -> D_verdade + lambda |G2|   */
            d_pen += calcula_d_penalizado(v, m) / 3.0;
        }
        atualiza_melhor_da_familia(&sel, configs.item[i], d_pen);
        if (d_pen < sel.vencedor.d_pen)        /* mais separavel */
            sel.vencedor = (Melhor){ configs.item[i], d_pen };
    }
    return sel;                        /* NAO treina classificador */
}

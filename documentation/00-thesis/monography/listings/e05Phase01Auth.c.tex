\begin{lstlisting}[language=C, caption={Fase 01 --- uma dobra externa do protocolo por dobra. Toda escolha usa só os locutores de treino da dobra; os de teste são pontuados uma única vez.}, label={lst:e05Phase01}]
/* treino: 12 locutores; teste: 3 locutores (disjuntos)          */
/* cand[]: os extratores da Fase 00 do sinal (voz 153, EEG 61)   */
/* grade[]: os 36 pontos (largura, blocos, taxa, decaimento)     */
/* As fontes fundidas leem as duas partes com o extrator da voz. */

Metricas dobra(Amostra treino[], Amostra teste[], int fonte,
               Combinacao cand[], int n_cand, Ponto grade[])
{
    /* 1. extrator: ordena os candidatos so com o treino          */
    int e = 0;
    for (int c = 0; c < n_cand; c++) {
        ajusta(cand[c], treino);             /* AE e PCA: treino */
        if (d_pen(cand[c], treino) < d_pen(cand[e], treino))
            e = c;
    }
    /* 2. hiperparametros: busca interna, 4 dobras por locutor    */
    Folds interno = divide_por_locutor(treino, 4);
    int p = menor_eer_medio(grade, 36, cand[e], fonte, interno);
                               /* empate: o de menos parametros  */
    /* 3. retreina o ponto escolhido sobre os 12 locutores        */
    Modelo m = treina_dsnn(grade[p], cand[e], fonte, treino);

    /* 4. uma unica avaliacao nos locutores de teste              */
    return avalia(m, cand[e], fonte, teste);   /* EER e AUC       */
}

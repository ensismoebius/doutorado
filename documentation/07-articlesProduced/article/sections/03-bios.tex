\section*{Biographies}
\label{sec:bio}

\begin{IEEEbiographynophoto}{André Furlan}
	(~\IEEEmembership{Member,~IEEE}) holds a Master’s degree in Computer Science from UNESP (Universidade Estadual Paulista), where he also taught. He currently teaches at CEETEPS (Paula Souza Center in São Paulo - Brazil) and conducts research in artificial intelligence, signal processing, and computational neuroscience. His experience includes systems development and applied research in brain–computer interfaces, EEG and voice signal analysis, and neuromorphic modeling using Spiking Neural Networks (SNNs). His academic background spans applied mathematics, programming, and bio-inspired computing. He is a reviewer for Springer journals and actively contributes to scientific dissemination and technical education in computing.
\end{IEEEbiographynophoto}

\begin{frame}
	\frametitle{Perda incompatível com a codificação: o que é?}
	\framesubtitle{O descompasso entre onde a informação está e o que a perda avalia}
	\small
	\par Em SNNs, a saída é uma \textbf{sequência de pulsos no tempo}. A função de perda deve mensurar a grandeza exata onde a informação foi codificada.
	\par Um \textbf{descompasso de codificação} ocorre quando a métrica ignora a dimensão temporal do código:
	\begin{center}
		\footnotesize
		\begin{tabular}{lcc}
			\toprule
			& \textbf{Alvo desejado} & \textbf{Previsão da rede} \\
			\midrule
			\textbf{Instante do pulso ($t$)} & $t_{\mathrm{alvo}} = 4$ & $t_{\mathrm{pred}} = 12$ \\
			\textbf{Contagem de pulsos ($N$)} & $N_{\mathrm{alvo}} = 1$ & $N_{\mathrm{pred}} = 1$ \\
			\midrule
			\textbf{Perda de contagem ($L_{\mathrm{taxa}}$)} & \multicolumn{2}{c}{$(N_{\mathrm{pred}} - N_{\mathrm{alvo}})^2 = (1 - 1)^2 = \mathbf{0} \implies \text{Erro zero (ilusão)}$} \\
			\textbf{Perda temporal ($L_{\mathrm{tempo}}$)} & \multicolumn{2}{c}{$(t_{\mathrm{pred}} - t_{\mathrm{alvo}})^2 = (12 - 4)^2 = \mathbf{64}$ (8 passos atrasado)} \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{itemize}
		\setlength{\itemsep}{1pt}
		\item A perda por contagem calcula gradiente zero e não corrige nada, embora o pulso tenha chegado completamente atrasado!
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Perda incompatível com a codificação: qual a sua importância?}
	\framesubtitle{Duas falhas silenciosas que sabotam o treinamento de SNNs}
	\footnotesize
	\begin{itemize}
		\setlength{\itemsep}{2pt}
		\item \textbf{Falha Silenciosa 1: Otimização cega por incompatibilidade}:
		\begin{itemize}
			\item As curvas de perda caem a zero e o otimizador reporta convergência, mas o modelo falha na tarefa porque a resposta temporal nunca foi ajustada.
		\end{itemize}
		\item \textbf{Falha Silenciosa 2: Gradientes mortos na perda temporal (\textit{SpikeTimeLoss})}:
		\begin{itemize}
			\item Para códigos temporais, mede-se a perda quadrática $L = (t_{\mathrm{pred}} - t_{\mathrm{alvo}})^2$.
			\item Se um neurônio \textbf{nunca dispara} em nenhum dos $T$ passos, adota-se a convenção de penalidade máxima $t_{\mathrm{pred}} = T$ (o fim da janela temporal).
			\item A perda reportada é alta, mas como o neurônio nunca cruzou o limiar, a derivada do tempo de disparo não fornece gradiente para erguer o potencial!
			\item O neurônio entra em silêncio irreversível durante o restante do treino.
		\end{itemize}
		\item \textbf{Diretriz}: Alinhar estritamente a perda ao código (taxa $\leftrightarrow$ contagem; latência $\leftrightarrow$ instante) e monitorar neurônios inativos.
	\end{itemize}
	\par\vfill{\tiny Referências: \cite{eshraghian2023training, liu2023firstspike}.}
\end{frame}
